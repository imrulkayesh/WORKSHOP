SELECT E.OFFICEMAIL,RO.STAFFID, E.OFFICEMOBILE,E.EMPLOYEENAME,M.*,MR.*,B.CENTERCODE FROM MT_PERTICIPANT P 
        INNER JOIN EM_EMPLOYEE E ON E.STAFFID=P.STAFFID
        INNER JOIN MT_MEETING M ON M.ID=P.MEETINGID
        INNER JOIN MT_MEETINGROOM MR ON MR.ID=M.MROOMID
        INNER JOIN BS_BUSINESSCENTER B ON B.ID=MR.BUSINESSCENTERID
        INNER JOIN MT_ROOMORGANIZER RO ON RO.STAFFID= M.APPROVERSTAFFID where RO.STAFFID = '95495'
        
        WHERE P.MEETINGID=138
